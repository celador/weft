import { randomBytes } from "node:crypto";

export type SessionOptions = { ttlMs: number };

export type Session = { token: string; userId: string; createdAt: number; expiresAt: number };

const sessions = new Map<string, Session>();

/** Platform default session lifetime, used when no options are given. */
export const DEFAULT_SESSION_OPTIONS: SessionOptions = { ttlMs: 60 * 60 * 1000 };

/** Start a session for a user and return it (the token is the bearer credential). */
export function createSession(userId: string, opts: SessionOptions): Session;
/** @deprecated Pass SessionOptions; kept for callers that use the platform default lifetime. */
export function createSession(userId: string): Session;
export function createSession(userId: string, opts: SessionOptions = DEFAULT_SESSION_OPTIONS): Session {
  const createdAt = Date.now();
  const session: Session = {
    token: randomBytes(24).toString("base64url"),
    userId,
    createdAt,
    expiresAt: createdAt + opts.ttlMs,
  };
  sessions.set(session.token, session);
  return session;
}

/** The live session for a token, or undefined (expired sessions are forgotten). */
export function getSession(token: string): Session | undefined {
  const session = sessions.get(token);
  if (session && Date.now() >= session.expiresAt) {
    sessions.delete(token);
    return undefined;
  }
  return session;
}

export function revokeSession(token: string): boolean {
  return sessions.delete(token);
}

/** Test helper: forget every session. */
export function resetSessions(): void {
  sessions.clear();
}

export type Item = { sku: string; price: number; qty: number };

/** calcTotal: sum of price × quantity over the cart. shippingFee: shipping for the cart, in dollars. */
export function calcTotal(items: Item[]): number {
  const subtotal = items.reduce((sum, item) => sum + item.price * item.qty, 0);
  const units = items.reduce((n, item) => n + item.qty, 0);
  return units >= 10 ? Math.round(subtotal * 90) / 100 : subtotal; // 10% bulk discount at 10+ units
}
export function shippingFee(items: Item[]): number { return calcTotal(items) >= 50 ? 0 : items.length ? 6 : 0; }
